注 ============================================================

"examples/分拣监控/页面" 名 页面根

匠 读页面源码(页名) 【
  判 【
    含(页名, "..") ⇒ 归 空
    它 ⇒ 空
  】
  判 【
    含(页名, "/") 或 含(页名, "\\") ⇒ 归 空
    它 ⇒ 空
  】
  页名 + ".页" 名 真名
  页面根 + "/" + 真名 名 路径
  判 【
    在(路径) == 假 ⇒ 归 空
    它 ⇒ 空
  】
  归 读(路径)
】

路("GET", "/api/页面", 节 (请) 【
  归 [
    { "名" ⇒ "总览", "文件" ⇒ "总览.页", "说明" ⇒ "监控大屏" },
    { "名" ⇒ "设备", "文件" ⇒ "设备.页", "说明" ⇒ "设备管理与控制" },
    { "名" ⇒ "分拣", "文件" ⇒ "分拣.页", "说明" ⇒ "分拣统计" },
    { "名" ⇒ "告警", "文件" ⇒ "告警.页", "说明" ⇒ "告警中心" },
    { "名" ⇒ "参数", "文件" ⇒ "参数.页", "说明" ⇒ "阈值设置" }
  ]
】)

路("GET", "/页/:页名", 节 (请) 【
  请.参["页名"] 名 页名
  读页面源码(页名) 名 源码
  判 【
    源码 == 空 ⇒ 归 应(404, { "Content-Type" ⇒ "text/plain; charset=utf-8" }, "没有这个页面：" + 页名)
    它 ⇒ 归 应(200, { "Content-Type" ⇒ "text/plain; charset=utf-8" }, 源码)
  】
】)

注 根路径返回总览页源码，方便浏览器直接打开 http://127.0.0.1:8795/
路("GET", "/", 节 (请) 【
  归 应(200, { "Content-Type" ⇒ "text/plain; charset=utf-8" }, 读页面源码("总览"))
】)
